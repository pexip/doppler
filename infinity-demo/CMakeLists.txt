# ---------------------------------------------------------------------------
# doppler-infinity - the Infinity-Client-REST-mode demo.
#
# Uses Pulse purely as a media engine; the Pexip Infinity Client REST API
# (https://docs.pexip.com/api_client/api_rest.htm) is driven from the app
# itself using libcurl. See infinity-demo/README.md for the flow.
#
# Self-contained, like pexninja/ and sip-demo/: the parent CMakeLists.txt
# just calls add_subdirectory(infinity-demo) when BUILD_DOPPLER_INFINITY=ON.
# ---------------------------------------------------------------------------

# ---- System dependencies --------------------------------------------------
# OpenGL / GLFW / Threads come from the parent's find_package() calls
# (they're cached). PkgConfig is used to find libcurl.
find_package(PkgConfig REQUIRED)

# libcurl is in the Ubuntu archive as `libcurl4-openssl-dev` (or the gnutls
# variant). The .pc file is named `libcurl`.
pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl)
message(STATUS "Found libcurl via pkg-config: ${CURL_VERSION}")

# ---- nlohmann/json (header-only, fetched at configure time) --------------
# Used by infinity_client.cpp to parse the small JSON envelopes returned
# by the Infinity REST endpoints (request_token, calls, refresh_token).
# Pinned to a tag so configure is reproducible. Tarball download (no clone)
# is plenty for a header-only lib and avoids dragging the test suite in.
include(FetchContent)
FetchContent_Declare(
    nlohmann_json
    URL      https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz
    URL_HASH SHA256=d6c65aca6b1ed68e7a182f4757257b107ae403032760ed6ef121c9d55e81757d)
# Skip the test suite + install rules - we just want the include path.
set(JSON_BuildTests        OFF CACHE INTERNAL "")
set(JSON_Install           OFF CACHE INTERNAL "")
set(JSON_MultipleHeaders   OFF CACHE INTERNAL "")
FetchContent_MakeAvailable(nlohmann_json)

# ---- Dear ImGui (reuses the parent's FetchContent declaration) -----------
# The parent CMakeLists.txt already calls FetchContent_MakeAvailable(imgui),
# so imgui_SOURCE_DIR is in scope here. We compile our own little imgui
# static lib so we don't tangle our backend choice with anyone else's.
if(NOT DEFINED imgui_SOURCE_DIR)
    message(FATAL_ERROR
        "imgui_SOURCE_DIR not set - infinity-demo/ must be added from the "
        "parent CMakeLists.txt after FetchContent_MakeAvailable(imgui).")
endif()

add_library(doppler_infinity_imgui STATIC
    ${imgui_SOURCE_DIR}/imgui.cpp
    ${imgui_SOURCE_DIR}/imgui_draw.cpp
    ${imgui_SOURCE_DIR}/imgui_tables.cpp
    ${imgui_SOURCE_DIR}/imgui_widgets.cpp
    ${imgui_SOURCE_DIR}/backends/imgui_impl_glfw.cpp
    ${imgui_SOURCE_DIR}/backends/imgui_impl_opengl3.cpp)
target_include_directories(doppler_infinity_imgui PUBLIC
    ${imgui_SOURCE_DIR}
    ${imgui_SOURCE_DIR}/backends)
target_link_libraries(doppler_infinity_imgui PUBLIC glfw OpenGL::GL)

# ---- The doppler-infinity executable -------------------------------------
add_executable(doppler-infinity
    src/main.cpp
    src/infinity_client.cpp
    src/udp_rtp_bridge.cpp)

target_link_libraries(doppler-infinity PRIVATE
    doppler_infinity_imgui
    pexip::pulse
    PkgConfig::CURL
    nlohmann_json::nlohmann_json
    Threads::Threads)

# Same RPATH treatment as the parent's `doppler` target so libpexpulse.so
# resolves without the user setting LD_LIBRARY_PATH for the direct dep.
set_target_properties(doppler-infinity PROPERTIES
    BUILD_RPATH   "${PEXIP_PREFIX}/lib"
    INSTALL_RPATH "${PEXIP_PREFIX}/lib")

# Convenience launcher (mirrors run-doppler.sh / run-doppler-sip.sh) -
# sets LD_LIBRARY_PATH so Pulse's transitive private siblings
# (libpexlgpl.so, libimf.so, ...) are resolvable at runtime.
file(GENERATE
    OUTPUT  "${CMAKE_BINARY_DIR}/run-doppler-infinity.sh"
    CONTENT "#!/bin/sh
# Auto-generated by CMake - sets LD_LIBRARY_PATH so libpexpulse.so's
# private siblings are resolvable, then execs the demo binary.
export LD_LIBRARY_PATH=\"${PEXIP_PREFIX}/lib:\${LD_LIBRARY_PATH}\"
exec \"$<TARGET_FILE:doppler-infinity>\" \"$@\"
"
    FILE_PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE
                     GROUP_READ GROUP_EXECUTE
                     WORLD_READ WORLD_EXECUTE)
